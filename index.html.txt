function buildInitialSlots() {

    const slots = [];

    for (let i = 1; i <= 50; i++) {

        let status = "available";

        // Slot 1 starts reserved, matching Recent Bookings above
        if (i === 1) status = "reserved";

        slots.push({
            number: i,
            label: "Slot " + i,
            type: SLOT_TYPES[(i - 1) % SLOT_TYPES.length],
            status: status
        });

    }

    return slots;

}


function logActivity(text) {

    const now = new Date();

    let hours = now.getHours();
    const minutes = now.getMinutes().toString().padStart(2, "0");
    const suffix = hours >= 12 ? "PM" : "AM";

    hours = hours % 12;
    if (hours === 0) hours = 12;

    state.activity.unshift({
        time: hours + ":" + minutes + " " + suffix,
        text: text
    });

    if (state.activity.length > 12) {
        state.activity.pop();
    }

}


function totalRevenue() {

    return state.payments.reduce((sum, p) => sum + p.amount, 0);

function renderRevenueBars() {

    const slots = [
        { label: "08:00 AM", value: 1200 },
        { label: "10:00 AM", value: 2100 },
        { label: "12:00 PM", value: 3400 },
        { label: "02:00 PM", value: 2700 + totalRevenue() }
    ];

    const max = Math.max(...slots.map(s => s.value));

    return slots.map(s => `
        <div class="bar-row">
            <div class="bar-label">${s.label}</div>
            <div class="bar-track">
                <div class="bar-fill" style="width:${Math.round((s.value / max) * 100)}%;"></div>
            </div>
            <div class="bar-value">₹${s.value.toLocaleString("en-IN")}</div>

function slotClicked(number) {

    const slot = state.slots.find(s => s.number === number);

    if (!slot) return;

    if (currentView !== "slots" && currentView !== "dashboard") return;

    // cycle status: available -> occupied -> reserved -> available
    const order = ["available", "occupied", "reserved"];
    const next = order[(order.indexOf(slot.status) + 1) % order.length];

    slot.status = next;

    logActivity("Slot A" + String(number).padStart(2, "0") + " marked " + next);

    showToast("Slot A" + String(number).padStart(2, "0") + " is now " + next);

    if (currentView === "slots") {
        showSlots();
    } else {
        showDashboard();
    }

}

function badgeClass(status) {

    if (status === "Confirmed") return "confirmed";
    if (status === "Pending") return "pending";
    if (status === "Cancelled") return "cancelled";
    return "pending";

}


function createBooking() {

    const user = document.getElementById("bkUser").value;
    const vehicle = document.getElementById("bkVehicle").value.trim().toUpperCase();
    const type = document.getElementById("bkType").value;
    const date = document.getElementById("bkDate").value;
    const start = document.getElementById("bkStart").value;
    const end = document.getElementById("bkEnd").value;

    if (!user || !vehicle || !date || !start || !end) {
        showToast("Please fill in all booking fields");
        return;
    }

    const bestSlot = state.slots.find(s => s.status === "available" && s.type === type)
        || state.slots.find(s => s.status === "available");

    if (!bestSlot) {
        showToast("No available slots right now");
        return;
    }

    bestSlot.status = "reserved";

    const booking = {
        id: nextBookingId++,
        user: user,
        vehicle: vehicle,
        vehicleType: type,
        slot: bestSlot.label,
        date: date,
        time: start + " - " + end,
        status: "Confirmed"
    };

    state.bookings.push(booking);

    logActivity("Booking confirmed for " + bestSlot.label);

    showToast("Best slot found: " + bestSlot.label + " reserved for " + vehicle);

    showBookings();

}


function cancelBooking(id) {

    const booking = state.bookings.find(b => b.id === id);

    if (!booking) return;

    booking.status = "Cancelled";

    const slot = state.slots.find(s => s.label === booking.slot);

    if (slot && slot.status === "reserved") {
        slot.status = "available";
    }

    logActivity("Booking " + id + " cancelled");

    showToast("Booking " + id + " cancelled");

    showBookings();

}

function processPayment() {

    const bookingIdRaw = document.getElementById("payBookingId").value.trim();
    const method = document.getElementById("payMethod").value;

    if (!bookingIdRaw) {
        showToast("Please enter a Booking ID");
        return;
    }

    const bookingId = parseInt(bookingIdRaw, 10);

    const booking = state.bookings.find(b => b.id === bookingId);

    if (!booking) {
        showToast("Booking ID not found");
        return;
    }

    const rateByType = { Car: 60, Bike: 30, EV: 90, Truck: 120 };
    const amount = rateByType[booking.vehicleType] || 60;

    const now = new Date();
    let hours = now.getHours();
    const minutes = now.getMinutes().toString().padStart(2, "0");
    const suffix = hours >= 12 ? "PM" : "AM";
    hours = hours % 12;
    if (hours === 0) hours = 12;

    state.payments.push({
        bookingId: bookingId,
        method: method,
        amount: amount,
        time: hours + ":" + minutes + " " + suffix
    });

    logActivity("Payment received ₹" + amount + " for Booking " + bookingId);

    showToast("Payment of ₹" + amount + " received via " + method);

    showPayments();

}
