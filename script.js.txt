function maxId(arr, key, fallback) { return arr.length ? Math.max(...arr.map(x => x[key])) : fallback; }
function makeSlots() {
  const slots = [];
  for (let i = 1; i <= 50; i++) {
    let type = "Car";
    if (i % 10 === 0) type = "EV"; else if (i % 5 === 0) type = "Bike";
    slots.push({ id: i, type, status: "available", floor: "Ground Floor" });
  }
  return slots;
}
function nowStr() { return new Date().toLocaleString(); }
function save() { localStorage.setItem("smartpark_state", JSON.stringify(state)); }
function load() { try { const raw = localStorage.getItem("smartpark_state"); return raw ? JSON.parse(raw) : null; } catch (e) { return null; } }
function log(text) { state.logs.unshift({ time: nowStr(), text }); state.logs = state.logs.slice(0, 40); }
function toast(msg) {
  const t = document.getElementById("toast");
  t.textContent = msg; t.classList.add("show");
  clearTimeout(toast._timer);
  toast._timer = setTimeout(() => t.classList.remove("show"), 2200);
}

function navigate(view) {

  document.getElementById("pageTitle").textContent = titles[view][0];

  document.getElementById("viewContent").innerHTML = map[view]();

}

function renderAnalytics() {
  const revenue = state.payments.reduce((s, p) => s + p.amount, 0);
  const byMethod = {}; state.payments.forEach(p => byMethod[p.method] = (byMethod[p.method] || 0) + p.amount);
  const maxMethod = Math.max(1, ...Object.values(byMethod));
  const byType = {}; state.bookings.forEach(b => byType[b.type] = (byType[b.type] || 0) + 1);
  const maxType = Math.max(1, ...Object.values(byType));
  const statusCounts = { available: 0, reserved: 0, occupied: 0 };
  state.slots.forEach(s => statusCounts[s.status]++);
  const occupancy = Math.round(((statusCounts.reserved + statusCounts.occupied) / state.slots.length) * 100);

  return `
    <div class="two-col">
      <div class="card"><h2>Revenue by Payment Method</h2>
        ${Object.keys(byMethod).length ? Object.entries(byMethod).map(([m, v]) => `<div class="bar-row"><div class="bar-label">${m}</div><div class="bar-track"><div class="bar-fill" style="width:${(v/maxMethod)*100}%"></div></div><div class="bar-value">₹${v}</div></div>`).join("") : `<p class="desc">No transactions yet</p>`}
      </div>
      <div class="card"><h2>Bookings by Vehicle Type</h2>
        ${Object.keys(byType).length ? Object.entries(byType).map(([t, v]) => `<div class="bar-row"><div class="bar-label">${t}</div><div class="bar-track"><div class="bar-fill" style="width:${(v/maxType)*100}%"></div></div><div class="bar-value">${v}</div></div>`).join("") : `<p class="desc">No bookings yet</p>`}
      </div>
    </div>
    <div class="card"><h2>System Performance</h2><div class="stat-grid">
      ${statCard("💰", "#22a35a", "Total Revenue", "₹" + revenue)}
      ${statCard("📅", "#2f6fed", "Total Bookings", state.bookings.length)}
      ${statCard("🚦", "#f6a623", "Occupancy Rate", occupancy + "%")}
      ${statCard("🅿️", "#22a35a", "Available Slots", statusCounts.available)}
    </div></div>`;
}

function attachHandlers(view) {
  if (view === "users") {
    document.getElementById("userForm").addEventListener("submit", e => {
      e.preventDefault();
      const name = document.getElementById("uName").value.trim();
      const phone = document.getElementById("uPhone").value.trim();
      const email = document.getElementById("uEmail").value.trim();
      if (!name || !phone || !email) return;
      state.users.push({ id: nextUserId++, name, phone, email, status: "Active" });
      log(`New user registered: ${name}`);
      save(); toast("User registered successfully"); navigate("users");
    });
  }
  if (view === "slots") {
    document.querySelectorAll(".slot").forEach(el => {
      el.addEventListener("click", () => {
        const slot = state.slots.find(s => s.id == el.dataset.slot);
        const order = ["available", "reserved", "occupied"];
        slot.status = order[(order.indexOf(slot.status) + 1) % order.length];
        save(); navigate("slots");
      });
    });
  }
  if (view === "bookings") {
    document.getElementById("bookingForm").addEventListener("submit", e => {
      e.preventDefault();
      const userId = Number(document.getElementById("bUser").value);
      const user = state.users.find(u => u.id === userId);
      const vehicle = document.getElementById("bVehicle").value.trim();
      const type = document.getElementById("bType").value;
      const date = document.getElementById("bDate").value;
      const start = document.getElementById("bStart").value;
      const end = document.getElementById("bEnd").value;
      if (!user || !vehicle || !date || !start || !end) { toast("Please fill all fields"); return; }
      let slot = state.slots.find(s => s.status === "available" && s.type === type);
      if (!slot) slot = state.slots.find(s => s.status === "available");
      if (!slot) { toast("No slots available right now"); return; }
      slot.status = "reserved";
      const booking = { id: nextBookingId++, userId: user.id, userName: user.name, vehicle, type, date, start, end, slotId: slot.id, status: "Confirmed", paid: false };
      state.bookings.push(booking);
      log(`Booking #${booking.id} created for ${user.name} — Slot ${slot.id} allocated`);
      save(); toast(`Slot ${slot.id} reserved — Booking #${booking.id}`); navigate("bookings");
    });
  }
  if (view === "payments") {
    document.getElementById("payForm").addEventListener("submit", e => {
      e.preventDefault();
      const bookingId = Number(document.getElementById("pBooking").value);
      const method = document.getElementById("pMethod").value;
      const booking = state.bookings.find(b => b.id === bookingId);
      if (!booking) { toast("Booking ID not found"); return; }
      if (booking.paid) { toast("This booking is already paid"); return; }
      const hours = Math.max(1, Math.round((toMinutes(booking.end) - toMinutes(booking.start)) / 60) || 1);
      const rate = RATE[booking.type] || 40;
      const amount = hours * rate;
      state.payments.push({ id: nextPaymentId++, bookingId, amount, method, date: booking.date });
      booking.paid = true;
      const slot = state.slots.find(s => s.id === booking.slotId);
      if (slot) slot.status = "occupied";
      log(`Payment of ₹${amount} received for Booking #${bookingId} via ${method}`);
      save(); toast(`Payment of ₹${amount} successful`); navigate("payments");
    });
  }
  if (view === "admin") {
    document.getElementById("btnBackup").addEventListener("click", () => { log("System backup triggered"); save(); toast("Backup completed"); navigate("admin"); });
    document.getElementById("btnExport").addEventListener("click", () => { log("Report exported"); toast("Report export started"); });
    document.getElementById("btnReset").addEventListener("click", () => {
      if (confirm("Reset all system data? This cannot be undone.")) { localStorage.removeItem("smartpark_state"); location.reload(); }
    });
  }
}

function toMinutes(hhmm) { if (!hhmm) return 0; const [h, m] = hhmm.split(":").map(Number); return h * 60 + m; }
function escapeHtml(str) { return String(str).replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;").replace(/"/g,"&quot;").replace(/'/g,"&#039;"); }
